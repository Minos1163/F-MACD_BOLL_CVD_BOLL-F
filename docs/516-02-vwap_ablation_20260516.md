# VWAP 门控消融令
**日期**: 2026-05-16  
**目标**: 消除 VWAP hard block 对同向确认信号（4H+1H 同向）的误杀  
**范围**: 仅消融 `same_direction` 路径，不触碰对冲/反向/shrink 路径  
**前提**: combo_hard_block 已上线，pre_live_assertions 必须通过

---

## 为什么 VWAP 在这里是反直觉的错误门控

```
当前行为（错误逻辑）：
  4H green_bar_growing + 1H green_bar_growing
  → MACD 双周期确认做空
  → 价格在下跌趋势中自然偏离 VWAP（dev = -6% ~ -13%）
  → vwap_hard_block 触发 → 总分归零 → HOLD

问题：
  价格大幅低于 VWAP 在下跌趋势中是正常现象
  VWAP 是日内均价参考，不是支撑/阻力
  "价格偏离 VWAP 太多" ≠ "不能做空"
  恰恰相反：强下跌趋势中，价格持续低于 VWAP 是空头动能强的证明

当前策略等价于：
  只允许在价格刚好在 VWAP 附近时做空（低偏离）
  但这往往是趋势中途短暂修复，不是最佳做空时机

正确逻辑：
  对于同向确认信号（4H + 1H 方向一致）：
    VWAP 偏离方向与交易方向一致 → 顺势，VWAP 不应阻断
    VWAP 偏离方向与交易方向相反 → 逆势追高/追低，VWAP 应阻断
```

## 消融设计（三层分级，不是全部删除）

```python
# src/fund_flow/macd_strategy_v2.py
# 替换 _check_vwap_gate() 函数

def _check_vwap_gate(
    self,
    price:       float,
    vwap:        float,
    atr_pct:     float,
    signal_4h:   str,
    signal_1h:   str,
    direction:   str,     # "long" or "short"
) -> dict:
    """
    VWAP 消融版门控：
    - 同向顺势（价格偏离与方向一致）：大幅放宽
    - 同向逆势（追高做多/追低做空）：保留保护
    - 对冲/方向不明：维持原有严格门控
    """
    if vwap <= 0:
        return {"action": "PASS", "mult": 1.0, "reason": "vwap_missing"}

    dev_pct = (price - vwap) / vwap          # 正 = 价格高于 VWAP

    # ── 判断偏离方向与交易方向的关系 ──────────────────────
    # 做空时：价格低于 VWAP (dev < 0) = 顺势偏离（空头趋势中正常）
    # 做多时：价格高于 VWAP (dev > 0) = 顺势偏离（多头趋势中正常）
    is_trend_aligned = (
        (direction == "short" and dev_pct < 0) or
        (direction == "long"  and dev_pct > 0)
    )
    is_counter_trend = (
        (direction == "short" and dev_pct > 0) or   # 做空但价格高于 VWAP
        (direction == "long"  and dev_pct < 0)       # 做多但价格低于 VWAP
    )

    # ── 判断方向确认强度 ────────────────────────────────────
    SAME_DIRECTION_PAIRS = {
        ("green_bar_growing",  "green_bar_growing"),   # 双周期空头加速
        ("green_bar_growing",  "green_bar_shrinking"),  # 4H空头+1H收缩
        ("red_bar_growing",    "red_bar_growing"),      # 双周期多头加速
        ("red_bar_growing",    "red_bar_shrinking"),    # 4H多头+1H收缩
    }
    is_same_direction = (signal_4h, signal_1h) in SAME_DIRECTION_PAIRS

    dev_abs = abs(dev_pct)

    # ════════════════════════════════════════════════════════
    # 路径 A：同向确认 + 顺势偏离（消融 VWAP 保护）
    # ════════════════════════════════════════════════════════
    if is_same_direction and is_trend_aligned:
        # 价格偏离是趋势本身造成的，不是追高追低
        # VWAP 在这里不是有效过滤器，消融

        if dev_abs < 0.05:          # dev < 5%：完全通过
            return {"action": "PASS", "mult": 1.0,
                    "reason": "same_dir_trend_aligned_pass"}

        if dev_abs < 0.10:          # dev 5%~10%：轻微减仓
            return {"action": "PENALTY", "mult": 0.90,
                    "reason": "same_dir_trend_aligned_penalty"}

        if dev_abs < 0.15:          # dev 10%~15%：probe 仓位
            return {"action": "PROBE", "mult": 0.80,
                    "max_portion": 0.06,
                    "reason": "same_dir_trend_aligned_probe"}

        # dev > 15%：极端偏离，仍阻断（防止极端行情追空）
        return {"action": "BLOCK", "mult": 0.0,
                "reason": "same_dir_extreme_deviation_block"}

    # ════════════════════════════════════════════════════════
    # 路径 B：同向确认 + 逆势偏离（追高做多/追低做空）
    # 保留保护，但从"hard block 3%"放宽到 ATR 归一化
    # ════════════════════════════════════════════════════════
    if is_same_direction and is_counter_trend:
        if atr_pct > 0:
            ratio = dev_abs / atr_pct
            if ratio < 2.0:
                return {"action": "PASS", "mult": 1.0,
                        "reason": "same_dir_counter_atr_pass"}
            if ratio < 3.5:
                return {"action": "PENALTY", "mult": 0.85,
                        "reason": "same_dir_counter_atr_penalty"}
            if ratio < 5.0:
                return {"action": "PROBE", "mult": 0.75,
                        "max_portion": 0.06,
                        "reason": "same_dir_counter_atr_probe"}
        else:
            # ATR 缺失退化：固定 3% 门槛
            if dev_abs < 0.03:
                return {"action": "PASS", "mult": 1.0,
                        "reason": "same_dir_counter_fallback_pass"}

        return {"action": "BLOCK", "mult": 0.0,
                "reason": "same_dir_counter_block"}

    # ════════════════════════════════════════════════════════
    # 路径 C：对冲/shrink/方向不明（维持原有严格门控）
    # ════════════════════════════════════════════════════════
    # 这里是 partial_confirm 候选和方向不明候选
    # 不消融，继续用 ATR 归一化

    if atr_pct > 0:
        ratio = dev_abs / atr_pct
        if ratio < 1.5:
            return {"action": "PASS",    "mult": 1.0,
                    "reason": "ambiguous_atr_pass"}
        if ratio < 2.5:
            return {"action": "PENALTY", "mult": 0.88,
                    "reason": "ambiguous_atr_penalty"}
        if ratio < 4.0:
            return {"action": "PROBE",   "mult": 0.80,
                    "max_portion": 0.06,
                    "reason": "ambiguous_atr_probe"}

    return {"action": "BLOCK", "mult": 0.0,
            "reason": "ambiguous_block"}
```

## 顺带修复：JST final 高分被 min_open 拦住

```
问题：JSTUSDT score=0.855 通过所有门控，到达 final，但 target=0.001 < min_open=0.06
根因：JST 价格极低（约 0.03 USDT），同样的 portion 换算成数量很大，
      但实际名义价值很小（portion=0.001 × 账户 ≈ 0.1 USDT），
      导致按名义价值计算后 portion 被极度压缩

正确修复：min_open_portion 检查应基于名义价值，而不是 portion 比例
```

```python
# src/app/fund_flow_bot.py — _apply_entry_gate_block()

def _check_min_open_portion(
    self,
    symbol:        str,
    target_portion: float,
    signal_score:  float,
    price:         float,
    account_equity: float,
) -> dict:
    """
    双重检查：
    1. portion 比例检查（当前逻辑）
    2. 名义价值检查（新增，防止 JST 类低价标的被误杀）
    """
    cfg = self.config

    # ── 检查 1：portion 比例 ────────────────────────────────
    min_open = cfg.min_open_portion   # 0.06

    # ── 检查 2：名义价值（新增）────────────────────────────
    notional = target_portion * account_equity
    min_notional_usdt = cfg.get("min_open_notional_usdt", 5.0)   # 最小 5 USDT

    if notional >= min_notional_usdt:
        # 名义价值足够，即使 portion 很小也允许开仓
        actual_portion = notional / account_equity
        return {"allow": True, "portion": actual_portion,
                "reason": "notional_check_passed"}

    # ── probe_floor_rescue 检查 ──────────────────────────────
    pf = cfg.probe_floor_rescue
    if pf.enabled and not pf.shadow_mode:
        if signal_score >= pf.min_score_threshold:    # 0.80
            return {"allow": True,
                    "portion": pf.probe_min_open_portion,   # 0.042
                    "reason": "probe_floor_rescue"}

    if pf.enabled and pf.shadow_mode:
        self._log_info(
            f"[PROBE_SHADOW] {symbol} score={signal_score:.4f} "
            f"eligible but shadow_mode=true"
        )

    return {"allow": False, "reason": "below_min_open_and_notional"}
```

# scripts/pre_live_assertions.py — 新增断言

def assert_vwap_ablation(cfg: dict) -> list[str]:
    errors = []
    vwap = cfg.get("vwap_deviation_gate", {})

    # 必须是新模式
    if vwap.get("mode") != "directional_ablation":
        errors.append(
            f"vwap_deviation_gate.mode={vwap.get('mode')}, "
            "must be directional_ablation"
        )

    # 旧字段必须删除
    if "vwap_deviation_hard_block" in cfg:
        errors.append(
            "vwap_deviation_hard_block still present, must be deleted"
        )

    # 顺势偏离门槛不能过严（防止消融被悄悄改回去）
    sad = vwap.get("same_dir_trend_aligned", {})
    if sad.get("pass_dev_pct", 0) < 0.04:
        errors.append(
            f"same_dir_trend_aligned.pass_dev_pct={sad.get('pass_dev_pct')} < 0.04, "
            "ablation ineffective"
        )

    return errors
