# Final 信号最小名义价值强制兜底
**日期**: 2026-05-18  
**问题**: final + score 过线的 SHORT，target 仍只有 0.000219，被 5x lift ratio 卡死  
**修复**: 去掉 lift ratio 限制，final 且过线的信号直接 lift 到 2~5 USDT，不改杠杆

---

## 根因一句话

```
仓位链输出 target=0.000219（灰尘级别）
risk_engine 需要提升 82x 才能到 2U
5x lift ratio 拒绝 → HOLD

修复：对于 stage=final + score>=threshold 的信号
      直接 target = max(target, min_notional/equity)
      不检查倍数，因为灰尘仓位的产生本身是上游 bug
      杠杆不变，只改 portion 比例
```

## [命令-1] 策略层输出处：final 信号加 notional floor

```python
# src/fund_flow/macd_strategy_v2.py
# _finalize_signal() 或等效的 signal 输出函数
# 在返回 final BUY/SELL decision 之前插入

def _apply_final_notional_floor(
    self,
    symbol:          str,
    direction:       str,
    target_portion:  float,
    signal_score:    float,
    threshold:       float,
    account_equity:  float,
    leverage:        int,
) -> float:
    """
    对已过 final + threshold 的信号，保证名义价值 >= min_notional。
    不改杠杆，只调整 portion。
    只在 signal_score >= threshold 时生效（即真正确认开仓的信号）。
    """
    # 只处理确认方向的信号（过线的），未过线的不 lift
    if signal_score < threshold:
        return target_portion

    min_notional = self._get_min_notional_usdt(symbol)   # 2U / 3U / 5U
    current_notional = target_portion * account_equity

    if current_notional >= min_notional:
        return target_portion   # 已经够大，不需要 lift

    # 计算 min portion（不改杠杆）
    min_portion = min_notional / account_equity
    # e.g. 2U / 112U = 0.01786

    # 检查不超过仓位上限
    max_portion = self.config.max_symbol_position_portion   # 0.35
    final_portion = min(min_portion, max_portion)

    self._log_info(
        f"[FINAL_FLOOR] {symbol} {direction} "
        f"score={signal_score:.4f} >= th={threshold:.4f} | "
        f"target={target_portion:.6f}({current_notional:.4f}U) → "
        f"floor={final_portion:.6f}({min_notional}U) | "
        f"leverage unchanged={leverage}x"
    )
    return final_portion


def _get_min_notional_usdt(self, symbol: str) -> float:
    """按品种返回最小名义价值"""
    cfg = self.config.min_open_notional
    if symbol == "BTCUSDT":
        return cfg.btc_usdt       # 5.0
    if symbol in cfg.major_symbols:
        return cfg.major_usdt     # 2.0（SOL/XRP/ETH 等）
    return cfg.default_usdt       # 2.0
```

## [命令-2] 同步修复 vol_vwap_warn 缩放不应作用于微仓

```python
# src/fund_flow/macd_strategy_v2.py 或 fund_flow_bot.py
# vol_vwap_warn_position_scale 的应用位置

def _apply_vol_vwap_warn_scale(
    self,
    target_portion:  float,
    vol_vwap_warn:   bool,
    scale:           float,          # 当前 0.5
    account_equity:  float,
    symbol:          str,
) -> float:
    """
    vol_vwap_warn 缩放只在 target 已经可执行时才生效。
    对不可执行（notional < 2U）的 target 不再二次压缩。
    """
    if not vol_vwap_warn:
        return target_portion

    min_notional = self._get_min_notional_usdt(symbol)
    current_notional = target_portion * account_equity

    if current_notional < min_notional:
        # target 本身不可执行，缩放没有风控意义，跳过
        self._log_debug(
            f"[VWAP_WARN_SKIP] notional={current_notional:.4f}U < {min_notional}U, "
            f"vol_vwap_warn scale skipped"
        )
        return target_portion

    # 可执行仓位才缩放
    scaled = target_portion * scale
    self._log_debug(
        f"[VWAP_WARN] {target_portion:.6f} × {scale} = {scaled:.6f}"
    )
    return scaled
```

## [命令-3] 去掉 risk_engine 的 lift ratio 限制

```python
# src/fund_flow/risk_engine.py
# validate_target_portion() 修改

def validate_target_portion(
    self,
    symbol:          str,
    direction:       str,
    target_portion:  float,
    signal_score:    float,
    threshold:       float,
    account_equity:  float,
    is_final:        bool = False,
) -> float | None:
    """
    final 且过线的信号：直接 lift 到 min_notional，不限 ratio。
    其余信号：维持原有逻辑。
    """
    notional     = target_portion * account_equity
    min_notional = self._get_min_notional(symbol)

    if notional >= min_notional:
        return target_portion   # 已经够，直接通过

    # final 且过线：无条件 lift（策略层理论上已经 lift，这里是双保险）
    if is_final and signal_score >= threshold:
        min_portion = min_notional / account_equity
        min_portion = min(min_portion, self.max_open_portion)
        self._log_info(
            f"[RISK_FLOOR] {symbol} {direction} "
            f"final+passed: lift {target_portion:.6f} → {min_portion:.6f} "
            f"({min_notional}U, ratio={min_portion/max(target_portion,1e-9):.0f}x)"
        )
        return min_portion

    # 非 final 或未过线：策略层 HOLD
    self._log_info(
        f"[RISK_HOLD] {symbol} notional={notional:.4f}U < {min_notional}U, HOLD"
    )
    return None
```
