  同一 symbol 被止损平仓后 1~2 个 cycle 又重新按同类方向开仓
  → 相当于用更高的价格重新承担同一个亏损

    self,

    score:      float,

            f"score={score:.4f} → shadow only"

        return {"action": "SHADOW", "max_portion": 0.0}

        self._log_info(
            f"[COMBO_PROBE] {symbol} {direction}+{signal_1h} "
            f"score={score:.4f} → forced probe max=0.042"
        )
        return {"action": "PROBE", "max_portion": 0.042}

    return {"action": "PASS", "max_portion": 9999}

## [命令-2] total_compression_floor 作用域修复

```python
# src/fund_flow/macd_strategy_v2.py 或 fund_flow_bot.py
# _calculate_position_size() 中 total_compression_floor 的应用

def _apply_total_compression_floor(
    self,
    score:       float,
    raw_portion: float,
    signal:      dict,
) -> float:
    """
    total_compression_floor 不允许覆盖以下情况：
    """
    cfg = self.config.total_compression_floor
    if not cfg.enabled:
        return raw_portion

    # ── 硬性豁免条件（这些情况下 floor 绝对不生效）──────────
    signal_1h = signal.get("signal_type_1h", "")

    FLOOR_EXEMPT_SIGNALS = {
        "flip_bearish", "green_bar_growing",    # 危险多头信号
        "flip_bullish", "red_bar_growing",      # 危险空头信号（如果做空）
    }

    is_trial     = signal.get("is_trial_entry", False)
    is_probe     = signal.get("is_probe", False)
    is_shrink    = signal_1h in {"red_bar_shrinking", "green_bar_shrinking"}
    is_dangerous = signal_1h in FLOOR_EXEMPT_SIGNALS

    if is_trial or is_probe or is_shrink or is_dangerous:
        # floor 不生效，保持 dynamic sizing 压缩结果
        self._log_debug(
            f"floor_exempt: trial={is_trial} probe={is_probe} "
            f"shrink={is_shrink} dangerous={is_dangerous}"
        )
        return raw_portion   # 不抬升

    # ── 正常路径：按 score 计算 floor ───────────────────────
    base = self.config.default_target_portion   # 0.35
    if score >= 0.85:
        floor = base * 0.45
    elif score >= 0.75:
        floor = base * 0.35
    elif score >= 0.65:
        floor = base * 0.25
    else:
        return raw_portion   # 低分不抬升

    return max(raw_portion, floor)
```

## [命令-3] ExitGuard 平仓后 cooldown

```python
# src/app/fund_flow_bot.py

class ExitCooldownRegistry:
    """
    记录 ExitGuard 触发的平仓事件，
    在 cooldown 期内禁止同 symbol 同方向再次开仓
    """
    def __init__(self, cooldown_seconds: int = 1800):  # 30 分钟
        self._cooldown_sec = cooldown_seconds
        self._exits: dict[str, dict] = {}   # symbol → {side, ts}

    def register_exit(self, symbol: str, side: str) -> None:
        self._exits[symbol] = {"side": side, "ts": time.time()}
        log_info(f"[COOLDOWN] {symbol} {side} exit registered, "
                 f"cooldown={self._cooldown_sec}s")

    def is_blocked(self, symbol: str, proposed_side: str) -> bool:
        entry = self._exits.get(symbol)
        if entry is None:
            return False

        # cooldown 是否过期
        elapsed = time.time() - entry["ts"]
        if elapsed > self._cooldown_sec:
            del self._exits[symbol]
            return False

        # 同方向才 block（不阻止反向对冲）
        if entry["side"] == proposed_side:
            log_info(
                f"[COOLDOWN] {symbol} {proposed_side} blocked, "
                f"elapsed={elapsed:.0f}s/{self._cooldown_sec}s"
            )
            return True

        return False


# 在 _process_symbol() 开仓决策前插入：
if self.exit_cooldown.is_blocked(symbol, proposed_direction):
    return DecisionResult.hold(reason="exit_cooldown_active")

# 在 ExitGuard 触发平仓后调用：
self.exit_cooldown.register_exit(symbol, closed_side)
```
