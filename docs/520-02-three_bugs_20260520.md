### 修复 A-2：below_vwap LONG 的风险阈值降到 -0.5%，并独立处理

```python
# 当前：vwap_dev < -1% 才算 below_vwap 风险
# 修复：降到 -0.5%，PUMP dev=-0.63% 才能被覆盖

# 同时新增：red_bar_growing + red_bar_growing 在 VWAP 下方的专项规则

def _apply_15m_entry_gate(
    self,
    direction:       str,
    raw_15m:         float,
    entry_15m:       str,
    vwap_dev_pct:    float,
    adx:             float,
    regime:          str,
    signal_4h:       str,
    signal_1h:       str,
    current_portion: float,
) -> dict:
    if direction != "long":
        return {"action": "PASS", "max_portion": current_portion}

    # ── 风险因子计数（阈值降低）──────────────────────────────
    is_below_vwap = vwap_dev_pct < -0.005   # 从 -1% 改为 -0.5%
    is_weak_adx   = adx < 20
    is_no_trade   = regime == "NO_TRADE"
    is_flip_entry = signal_1h in ("flip_bullish",) or signal_4h in ("flip_bullish",)

    # 新增：red_bar_growing 双向共振在 VWAP 下方 = 独立高风险
    is_rbg_below_vwap = (
        is_below_vwap
        and signal_4h == "red_bar_growing"
        and signal_1h == "red_bar_growing"
    )

    risk_count = sum([is_below_vwap, is_weak_adx, is_no_trade, is_flip_entry])

    # ── red_bar_growing 双向 + VWAP 下方：独立处理 ───────────
    if is_rbg_below_vwap:
        has_15m = raw_15m >= 0.30 and entry_15m != "-"
        if not has_15m:
            # 没有 15M 确认：最多 probe
            return {
                "action":      "PROBE",
                "max_portion": 0.060,
                "reason": (
                    f"rbg_below_vwap dev={vwap_dev_pct:.2%} "
                    f"raw_15m={raw_15m:.2f} no_confirm → probe 0.06"
                ),
            }
        # 有 15M 但仍在 VWAP 下方：允许通过，压仓
        return {
            "action":      "PASS",
            "max_portion": min(current_portion, 0.100),
            "reason": f"rbg_below_vwap with_15m → cap 0.10",
        }

    if risk_count == 0:
        return {"action": "PASS", "max_portion": current_portion}

    has_strong_15m = raw_15m >= 0.40 and entry_15m != "-"
    has_medium_15m = raw_15m >= 0.30
    has_weak_15m   = raw_15m >= 0.20

    if risk_count >= 3:
        if not has_medium_15m:
            return {"action": "BLOCK",
                    "reason": f"multi_risk({risk_count}) no_medium_15m → block"}
        if not has_strong_15m:
            return {"action": "PROBE", "max_portion": 0.042,
                    "reason": f"multi_risk({risk_count}) medium_15m → probe 0.042"}
        return {"action": "PROBE", "max_portion": 0.060,
                "reason": f"multi_risk({risk_count}) strong_15m → probe 0.060"}

    if risk_count == 2:
        if not has_medium_15m:
            return {"action": "PROBE", "max_portion": 0.042,
                    "reason": f"dual_risk weak_15m={raw_15m:.2f} → probe 0.042"}
        if not has_strong_15m:
            return {"action": "PROBE", "max_portion": 0.080,
                    "reason": "dual_risk medium_15m → probe 0.080"}
        return {"action": "PASS", "max_portion": min(current_portion, 0.100),
                "reason": "dual_risk strong_15m → cap 0.10"}

    # risk_count == 1
    if not has_weak_15m:
        return {"action": "PROBE", "max_portion": 0.060,
                "reason": f"single_risk very_weak_15m={raw_15m:.2f} → probe 0.060"}
    return {"action": "PASS", "max_portion": min(current_portion, 0.160),
            "reason": "single_risk ok_15m → cap 0.16"}
```

### 修复 A-3：final floor lift 不覆盖 gate cap

```python
# src/fund_flow/macd_strategy_v2.py
# _apply_final_notional_floor() 修改

def _apply_final_notional_floor(
    self,
    symbol:           str,
    direction:        str,
    target_portion:   float,
    signal_score:     float,
    threshold:        float,
    account_equity:   float,
    leverage:         int,
    gate_cap_applied: bool,     # 新增参数：是否已经被 gate 压缩过
    gate_cap_portion: float,    # 新增参数：gate 的上限 portion
) -> float:
    if signal_score < threshold:
        return target_portion

    min_notional  = self._get_min_notional_usdt(symbol)
    current_notional = target_portion * account_equity

    if current_notional >= min_notional:
        return target_portion

    min_portion = min_notional / account_equity

    # 关键修复：floor lift 不能超过 gate 设定的 cap
    if gate_cap_applied:
        # 只提升到 min_notional，但不超过 gate_cap
        final_portion = min(min_portion, gate_cap_portion)
        self._log_info(
            f"[FINAL_FLOOR] {symbol} lift {target_portion:.6f} → "
            f"{final_portion:.6f} (gate_cap={gate_cap_portion:.4f})"
        )
        return final_portion

    # 没有 gate cap：正常 lift 到 min_notional
    final_portion = min(min_portion, self.config.max_symbol_position_portion)
    return final_portion
```

### 修复 B-1：capacity 检查改为 bucket 优先

```python
# src/fund_flow/decision_engine.py 或 fund_flow_bot.py
# _check_active_symbol_capacity() 替换

def _check_active_symbol_capacity(
    self,
    symbol:             str,
    proposed_notional:  float,     # 本次开仓的名义价值（USDT）
    existing_positions: list[dict],
) -> tuple[bool, str]:
    """
    bucket 优先容量检查：
      小仓位（<5U）：最多 5 个
      大仓位（>=5U）：最多 4 个
      总计最多 9 个
    返回 (allowed, reason)
    """
    cfg = self.config.position_count_limit_by_margin

    if not cfg.get("enabled", False):
        # 旧逻辑兜底
        total = len(existing_positions)
        max_t = self.config.max_active_symbols
        if total >= max_t:
            return False, f"legacy_cap threshold={max_t} value={total}"
        return True, ""

    small_thresh = cfg["small_margin_threshold_usdt"]   # 5.0
    max_small    = cfg["max_small_margin_positions"]     # 5
    max_large    = cfg["max_large_margin_positions"]     # 4
    max_total    = max_small + max_large                 # 9

    # 统计现有仓位
    small_count = sum(1 for p in existing_positions
                      if p.get("notional_usdt", 0) < small_thresh)
    large_count = sum(1 for p in existing_positions
                      if p.get("notional_usdt", 0) >= small_thresh)
    total_count = len(existing_positions)

    # 判断本次开仓属于哪个 bucket
    is_small = proposed_notional < small_thresh

    if total_count >= max_total:
        return False, f"total_cap threshold={max_total} value={total_count}"

    if is_small and small_count >= max_small:
        return False, f"small_bucket_cap threshold={max_small} value={small_count}"

    if not is_small and large_count >= max_large:
        return False, f"large_bucket_cap threshold={max_large} value={large_count}"

    return True, ""
```

## Bug C：动态杠杆统一为 3/4/5

### 根因

```python
# 内部 2x cap 的位置（需要全部改）：
caps_to_fix = {
    "probe_leverage_cap":                   2,  # → 3
    "preflip_trial_max_leverage":           2,  # → 3
    "red_bar_growing_probe_max_leverage":   2,  # → 3
    "green_bar_growing_probe_max_leverage": 2,  # → 3
    "flip_bearish_normal_boll_max_leverage": 2, # → 3

# calculate_leverage 的阶梯（推断）：
# score >= 0.85 → 4x → 被 probe cap 2x → 最终 clamp 到 min=3 → 3x
# 正确应该是：probe cap 3x → 不被 clamp 改变 → 3x（probe）
# 高分非 probe → 4x 或 5x

# 修复：统一 leverage ladder
def calculate_leverage(
    self,
    signal_score:   float,
    is_probe:       bool,
    is_trial:       bool,
    direction:      str,
) -> int:
    """
    动态杠杆 3/4/5 阶梯：
      probe / trial                          → 3x（最保守）
      normal, score < 0.75                   → 3x
      normal, 0.75 <= score < 0.85           → 4x
      normal, score >= 0.85                  → 5x
    """
    MIN_LEV = self.config.min_leverage       # 3
    MAX_LEV = self.config.max_leverage       # 5

    # probe/trial 固定最低杠杆
    if is_probe or is_trial:
        return MIN_LEV   # 3x

    # 按分数阶梯
    if signal_score >= 0.85:
        lev = 5
    elif signal_score >= 0.75:
        lev = 4
    else:
        lev = 3

    # 限制在 [min, max] 区间
    return max(MIN_LEV, min(lev, MAX_LEV))
```
