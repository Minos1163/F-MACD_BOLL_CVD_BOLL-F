## 问题诊断

```
当前 RSI hard veto 的语义（错误的）：
  green_bar_growing + green_bar_shrinking → RSI against → 总分归零

这个判断把"1H 空头动能收缩"当成了"1H 方向与 4H 相反"
但 green_bar_shrinking 不是反向信号，是同向趋势内的动能减弱

正确语义应该是：
  4H green_bar_growing（空头加速）
  + 1H green_bar_shrinking（空头收缩 = 短暂回调/修复）
  = 趋势中继/回调买卖点，是有效 SHORT 候选

等价的多头类比：
  4H red_bar_growing（多头加速）
  + 1H red_bar_shrinking（多头收缩 = 正常回调）
  = 这是教科书的"趋势回调做多"入场点
  RSI 不应该在这里 hard veto

本窗口量化损失：
  170 个 green_bar_growing + green_bar_shrinking
  其中 147 个被 rsi_1h_direction_against_veto 清零
  这 147 个如果有 60% WR，预期应产生约 88 笔有效候选
  但实际产生 0 笔（其中 1 笔是 JST 用其他路径通过的）
```

    signal_1h:  str,

### 1-B 替换执行逻辑

```python
def _evaluate_rsi_direction_gate(
    self,
    signal_4h:       str,
    signal_1h:       str,
    direction:       str,
    rsi_direction_1h: str,    # "up" / "down" / "flat"
    rsi_score:       float,
    signal_score_so_far: float,
) -> dict:
    """
    替换原有的 hard veto 逻辑
    返回：{ "action": "PASS"|"PENALTY"|"VETO", "mult": float, "reason": str }
    """
    gate_type = _get_rsi_direction_gate_type(signal_4h, signal_1h, direction)

    # ── PASS：方向完全确认，RSI 正常流程 ────────────────────
    if gate_type == "PASS":
        return {"action": "PASS", "mult": 1.0,
                "reason": "trend_confirmed_pass"}

    # ── 判断 RSI 是否真的反向 ────────────────────────────────
    expected_rsi_dir = "down" if direction == "short" else "up"
    rsi_is_against = (rsi_direction_1h not in (expected_rsi_dir, "flat"))
    rsi_is_flat    = (rsi_direction_1h == "flat")

    # ── SOFT：趋势中继，RSI against → penalty（不再 hard veto）
    if gate_type == "SOFT":
        if rsi_is_against:
            # 趋势中继中 RSI 短暂反向 = 回调，打折但不归零
            return {
                "action":  "PENALTY",
                "mult":    0.88,
                "max_portion": 0.06,    # 强制 probe 仓位
                "reason":  f"trend_continuation_{signal_4h}+{signal_1h}_rsi_against_soft",
            }
        if rsi_is_flat:
            # RSI 走平 = 动能不足，轻微打折
            return {
                "action": "PENALTY",
                "mult":   0.93,
                "reason": f"trend_continuation_{signal_4h}+{signal_1h}_rsi_flat_soft",
            }
        # RSI 同向：正常通过
        return {"action": "PASS", "mult": 1.0,
                "reason": "trend_continuation_rsi_aligned"}

    # ── HARD：真正的方向冲突，维持 veto ──────────────────────
    if rsi_is_against:
        return {
            "action": "VETO",
            "mult":   0.0,
            "reason": f"direction_conflict_{signal_4h}+{signal_1h}_rsi_hard_veto",
        }
    if rsi_is_flat:
        return {
            "action": "PENALTY",
            "mult":   0.85,
            "reason": f"direction_weak_{signal_4h}+{signal_1h}_rsi_flat_hard",
        }
    return {"action": "PASS", "mult": 1.0, "reason": "hard_combo_rsi_aligned"}
```

## [命令-3] Probe 名义价值冲突修复

```python
# 问题量化：
# probe_min_open = 0.042 （= 4.2% 仓位）
# min_open_notional_usdt = 5.0 USDT
# account_equity = 111.18 USDT
# probe notional = 0.042 × 111.18 = 4.67 USDT < 5.0 → 被拦住

# 这是两个门槛互相打架：
# probe floor 说"4.2% 可以"，名义价值说"4.67U 不行"
# 结果是 probe floor 永远无法在 < 119U 的账户上生效

# 修复方案 A（推荐）：min_open_notional 与 probe_min_open 联动
# min_open_notional = probe_min_open × account_equity × 0.95（留 5% 余量）
# 这样 probe floor 允许的最小仓位永远能通过 notional 检查

# 修复方案 B（备选）：固定 min_open_notional 为账户的百分比
# min_open_notional = account_equity × 0.040（= 4%，低于 probe 的 4.2%）

def _resolve_min_open_notional(
    self,
    account_equity: float,
) -> float:
    """
    动态计算 min_open_notional，确保不高于 probe_min_open 对应的名义价值
    """
    probe_min_portion = self.config.probe_floor_rescue.probe_min_open_portion
    # probe 对应的名义价值，再乘 0.95 留安全余量
    probe_notional = probe_min_portion * account_equity * 0.95

    # 与配置值取较小者（不能比配置值更宽松）
    config_notional = self.config.get("min_open_notional_usdt", 5.0)

    return min(probe_notional, config_notional)


# 在 _check_min_open_portion() 中使用动态值：
def _check_min_open_portion(
    self,
    symbol:         str,
    target_portion: float,
    signal_score:   float,
    price:          float,
    account_equity: float,
) -> dict:
    notional = target_portion * account_equity
    min_notional = self._resolve_min_open_notional(account_equity)

    if notional >= min_notional:
        return {"allow": True, "portion": target_portion,
                "reason": "notional_check_passed"}

    # probe rescue
    pf = self.config.probe_floor_rescue
    if pf.enabled and not pf.shadow_mode:
        if signal_score >= pf.min_score_threshold:
            probe_notional = pf.probe_min_open_portion * account_equity
            if probe_notional >= min_notional:
                return {"allow": True,
                        "portion": pf.probe_min_open_portion,
                        "reason":  "probe_floor_rescue"}

    return {"allow": False,
            "reason": f"notional={notional:.3f} < min={min_notional:.3f}"}
```

def assert_rsi_adaptive_gate(cfg: dict) -> list[str]:
    errors = []
    rsi = cfg.get("rsi_rhythm", {})
    adg = rsi.get("adaptive_direction_gate", {})

            f"soft_rsi_against_max_portion={adg.get('soft_rsi_against_max_portion')}"
            " > 0.10, probe cap too loose"
        )

    return errors


def assert_probe_notional_coherence(cfg: dict) -> list[str]:

    probe_portion = (cfg.get("probe_floor_rescue", {})
                        .get("probe_min_open_portion", 0.042))
