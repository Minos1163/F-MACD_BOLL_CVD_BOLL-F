# 15M Beta 风控加速退出
**日期**: 2026-05-22  
**目标**: 把 4~18 bars 的 ExitGuard 确认，在 BTC+单币 15M 同步反向时压缩到 1~2 bars  
**设计原则**: BTC 不替代单币方向，只加速确认；低相关性单币自动关闭 BTC 权重

---

## 核心数据支撑

```
48H 亏损 entry 23 个：
  22/23 的 15M raw < 0.30 或 entry15m = "-"（入场时短周期就弱）
  FET 持仓 465min，ICP SHORT 持仓 30min 后亏损
  ExitGuard 确认时间：XRP 18 bars, FET 10 bars, TRUMP 4 bars, XLM 5 bars

目标：开仓后前 30min（2 根 15M），若 BTC+单币同步反向 → 1 bar full close
      开仓后 30~60min，BTC 领先单币跟随 → 2 bars reduce 50%
      低相关性单币（corr < 0.20）→ 不用 BTC，只看单币 15M
```

## [命令-1] BTC Beta 风控评分器

```python
# src/fund_flow/btc_beta_risk.py（新文件）

from dataclasses import dataclass, field
from collections import deque
import numpy as np

@dataclass
class BtcBetaRiskConfig:
    enabled:                     bool  = True
    # 相关性阈值：低于此值不使用 BTC 权重
    min_corr_for_btc_weight:    float = 0.20
    # BTC 反向阈值（15M 收益率）
    btc_against_long_pct:       float = -0.0015    # BTC 15M 跌 0.15% 算反向
    btc_against_short_pct:      float = +0.0015    # BTC 15M 涨 0.15% 算反向
    # 单币反向阈值
    alt_against_15m_pct:        float = -0.0010    # 单币 15M 跌 0.10% 算反向
    alt_against_30m_pct:        float = -0.0015    # 30M 阈值略高
    # 滚动相关性窗口
    corr_window_bars:           int   = 48          # 12H = 48 根 15M
    # fast-fail 窗口
    fast_fail_window_bars:      int   = 2           # 开仓后前 2 根 15M
    fast_fail_mae_threshold:    float = -0.002      # fast-fail 的 MAE 上限
    fast_fail_mfe_threshold:    float = +0.002      # fast-fail 的 MFE 下限（没兑现）


class BtcBetaRiskScorer:
    """
    计算每个持仓的 BTC beta 风控得分。
    得分越高 = 当前需要越紧迫地退出。
    """

    def __init__(self, config: BtcBetaRiskConfig):
        self.cfg = config
        # symbol → deque of (btc_ret_15m, alt_ret_15m)
        self._corr_history: dict[str, deque] = {}

    # ── 相关性维护 ───────────────────────────────────────────────

    def update_corr_history(
        self,
        symbol:        str,
        btc_ret_15m:   float,
        alt_ret_15m:   float,
    ) -> None:
        """每根 15M K 线结束后调用，更新滚动相关性历史"""
        if symbol not in self._corr_history:
            self._corr_history[symbol] = deque(maxlen=self.cfg.corr_window_bars)
        self._corr_history[symbol].append((btc_ret_15m, alt_ret_15m))

    def get_btc_alt_corr(self, symbol: str) -> float:
        """计算 BTC 与 symbol 的同步相关性（lag=0）"""
        hist = self._corr_history.get(symbol, [])
        if len(hist) < 10:
            return 0.0   # 样本不足：返回 0（不使用 BTC 权重）
        btc = [h[0] for h in hist]
        alt = [h[1] for h in hist]
        try:
            return float(np.corrcoef(btc, alt)[0, 1])
        except Exception:
            return 0.0

    # ── 风控评分 ─────────────────────────────────────────────────

    def score(
        self,
        symbol:          str,
        direction:       str,        # "long" or "short"
        btc_ret_15m:     float,      # BTC 最新 15M 收益率
        btc_ret_30m:     float,      # BTC 最近 2 根 15M 累计收益率
        alt_ret_15m:     float,      # 单币最新 15M 收益率
        alt_ret_30m:     float,      # 单币最近 30M 累计收益率
        position_age_bars: int,      # 持仓已过多少根 15M K 线
        mfe_pct:         float,      # 最大浮盈（正数）
        mae_pct:         float,      # 最大浮亏（负数）
    ) -> dict:
        """
        返回风控评分字典：
          risk_score:  int        0~5，越高越紧迫
          action:      str        "HOLD" | "REDUCE_50" | "CLOSE"
          reason:      str
          use_btc:     bool       是否使用了 BTC 权重
        """
        cfg = self.cfg
        if not cfg.enabled:
            return {"risk_score": 0, "action": "HOLD", "reason": "disabled"}

        corr = self.get_btc_alt_corr(symbol)
        use_btc = corr >= cfg.min_corr_for_btc_weight

        risk_score = 0
        reasons    = []

        # ── 单币 15M 反向（始终计算，不受 BTC 相关性影响）───────
        if direction == "long":
            alt_against_15m = alt_ret_15m <= cfg.alt_against_15m_pct
            alt_against_30m = alt_ret_30m <= cfg.alt_against_30m_pct
        else:
            alt_against_15m = alt_ret_15m >= abs(cfg.alt_against_15m_pct)
            alt_against_30m = alt_ret_30m >= abs(cfg.alt_against_30m_pct)

        if alt_against_15m:
            risk_score += 1
            reasons.append(f"alt_15m_against({alt_ret_15m:.3%})")

        if alt_against_30m:
            risk_score += 1
            reasons.append(f"alt_30m_against({alt_ret_30m:.3%})")

        # ── BTC 反向（仅高相关性时计算）──────────────────────────
        if use_btc:
            if direction == "long":
                btc_against_15m = btc_ret_15m <= cfg.btc_against_long_pct
                btc_against_30m = btc_ret_30m <= cfg.btc_against_long_pct * 1.5
            else:
                btc_against_15m = btc_ret_15m >= cfg.btc_against_short_pct
                btc_against_30m = btc_ret_30m >= cfg.btc_against_short_pct * 1.5

            if btc_against_15m:
                risk_score += 1
                reasons.append(f"btc_15m_against({btc_ret_15m:.3%},corr={corr:.2f})")

            if btc_against_30m and alt_against_30m:
                # BTC 与单币 30M 同步反向：额外加权（这是最强信号）
                risk_score += 1
                reasons.append(f"btc_alt_30m_sync({btc_ret_30m:.3%},{alt_ret_30m:.3%})")

        # ── Fast-fail 窗口（开仓前 2 根 15M）─────────────────────
        in_fast_fail = position_age_bars <= cfg.fast_fail_window_bars
        no_mfe_gain  = mfe_pct < cfg.fast_fail_mfe_threshold
        has_mae      = mae_pct <= cfg.fast_fail_mae_threshold

        if in_fast_fail and (alt_against_15m or (use_btc and btc_against_15m)):
            if no_mfe_gain and has_mae:
                risk_score += 2   # 快速失败：分数激增
                reasons.append(
                    f"fast_fail(age={position_age_bars}bars,"
                    f"mfe={mfe_pct:.3%},mae={mae_pct:.3%})"
                )
            elif no_mfe_gain:
                risk_score += 1
                reasons.append(f"fast_fail_no_mfe(age={position_age_bars}bars)")

        # ── 决策映射 ──────────────────────────────────────────────
        reason_str = " | ".join(reasons) if reasons else "no_risk"

        if risk_score >= 4:
            return {"risk_score": risk_score, "action": "CLOSE",
                    "reason": reason_str, "use_btc": use_btc}

        if risk_score >= 2:
            return {"risk_score": risk_score, "action": "REDUCE_50",
                    "reason": reason_str, "use_btc": use_btc}

        if risk_score == 1 and in_fast_fail:
            return {"risk_score": risk_score, "action": "REDUCE_50",
                    "reason": reason_str + " [fast_fail_window]", "use_btc": use_btc}

        return {"risk_score": risk_score, "action": "HOLD",
                "reason": reason_str, "use_btc": use_btc}
```

## [命令-2] 集成到 ExitGuard

```python
# src/app/fund_flow_bot.py
# _apply_exit_signal_guard() 内，在 reverse_bars 计数前插入

def _run_exit_cycle(self, symbol: str, position: dict) -> None:
    """
    每个 cycle 对持仓运行两层退出检查：
    Layer 1: BtcBetaRiskScorer（15M beta 加速退出）
    Layer 2: ExitSignalGuard  （原有多 bars 确认退出）
    Layer 1 优先级高于 Layer 2
    """

    # ── 获取最新 15M 数据 ─────────────────────────────────────
    btc_15m  = self._get_btc_ret_15m()    # BTC 最新 15M 收益率
    btc_30m  = self._get_btc_ret_30m()
    alt_15m  = self._get_alt_ret_15m(symbol)
    alt_30m  = self._get_alt_ret_30m(symbol)

    # 维护相关性历史（每 cycle 调用一次）
    self.btc_beta_scorer.update_corr_history(symbol, btc_15m, alt_15m)

    # ── Layer 1: BTC Beta 风控 ────────────────────────────────
    age_bars = position.get("hold_bars_15m", 0)   # 已持仓 15M K 线数
    mfe_pct  = position.get("mfe_pct", 0.0)
    mae_pct  = position.get("mae_pct", 0.0)

    beta_result = self.btc_beta_scorer.score(
        symbol         = symbol,
        direction      = position["side"].lower(),
        btc_ret_15m    = btc_15m,
        btc_ret_30m    = btc_30m,
        alt_ret_15m    = alt_15m,
        alt_ret_30m    = alt_30m,
        position_age_bars = age_bars,
        mfe_pct        = mfe_pct,
        mae_pct        = mae_pct,
    )

    self._log_debug(
        f"[BETA_RISK] {symbol} score={beta_result['risk_score']} "
        f"action={beta_result['action']} "
        f"use_btc={beta_result['use_btc']} "
        f"reason={beta_result['reason']}"
    )

    # ── Layer 1 触发退出 ─────────────────────────────────────
    if beta_result["action"] == "CLOSE":
        self._log_info(
            f"[BETA_CLOSE] {symbol} risk_score={beta_result['risk_score']} "
            f"→ full close | {beta_result['reason']}"
        )
        self._execute_exit_guard_close(symbol, ratio=1.0,
                                        reason=f"beta_risk_close: {beta_result['reason']}")
        return   # 已退出，跳过 Layer 2

    if beta_result["action"] == "REDUCE_50":
        self._log_info(
            f"[BETA_REDUCE] {symbol} risk_score={beta_result['risk_score']} "
            f"→ reduce 50% | {beta_result['reason']}"
        )
        self._execute_exit_guard_close(symbol, ratio=0.50,
                                        reason=f"beta_risk_reduce: {beta_result['reason']}")
        # 继续运行 Layer 2 对剩余 50% 仓位

    # ── Layer 2: 原有 ExitSignalGuard（不变） ─────────────────
    self._apply_exit_signal_guard_original(symbol, position)
```

## [命令-3] 15M 数据采集

```python
# src/fund_flow/btc_beta_risk.py（续）

class BtcPriceCache:
    """
    轻量级 BTC 15M 价格缓存，每 cycle 更新一次。
    不依赖额外 API 调用，复用 MarketIngestionService 的数据。
    """

    def __init__(self, max_bars: int = 6):
        self._prices: deque = deque(maxlen=max_bars)  # 最近 6 根 15M 收盘价

    def push(self, close_price: float) -> None:
        self._prices.append(close_price)

    def ret_15m(self) -> float:
        """最新 15M 收益率"""
        if len(self._prices) < 2:
            return 0.0
        return (self._prices[-1] - self._prices[-2]) / self._prices[-2]

    def ret_30m(self) -> float:
        """最近 30M 累计收益率"""
        if len(self._prices) < 3:
            return 0.0
        return (self._prices[-1] - self._prices[-3]) / self._prices[-3]


# 在 fund_flow_bot.py 的 MarketIngestionService 里，每 cycle 调用：
def _update_btc_price_cache(self) -> None:
    btc_snapshot = self.market_ingestion.get_symbol_snapshot("BTCUSDT")
    if btc_snapshot:
        self.btc_price_cache.push(btc_snapshot.close_price)

def _get_btc_ret_15m(self) -> float:
    return self.btc_price_cache.ret_15m()

def _get_btc_ret_30m(self) -> float:
    return self.btc_price_cache.ret_30m()

def _get_alt_ret_15m(self, symbol: str) -> float:
    snap = self.market_ingestion.get_symbol_snapshot(symbol)
    # 从 symbol 的 15M OHLC 数据计算
    return snap.ret_15m if snap else 0.0
```
