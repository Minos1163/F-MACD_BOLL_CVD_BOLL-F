# src/app/fund_flow_bot.py — 在 __init__ 或 run() 最开始插入
# 不是"建议加"，是"必须加，否则每次都在猜"

import hashlib, json, time

def _print_config_fingerprint(self) -> None:
    """
    启动时和每隔 N 个 cycle 打印关键参数快照
    让日志可以直接验证部署是否生效，不靠猜
    """
    cfg = self.config

    # 提取关键参数
    snapshot = {
        "ts_bj":            time.strftime("%Y-%m-%d %H:%M:%S BJ"),
        "probe_shadow":     cfg.probe_floor_rescue.shadow_mode,
        "probe_min_open":   cfg.get("min_open_portion_probe", "MISSING"),
        "weight_rsi":       cfg.scoring_weights.weight_rsi_rhythm,
        "weight_4h":        cfg.scoring_weights.weight_4h_direction,
        "weight_4h_enh":    cfg.scoring_weights.weight_4h_enhancement,
        "vwap_gate_mode":   cfg.get("vwap_deviation_gate", {}).get("mode", "LEGACY"),
        "vwap_hard_block":  cfg.get("vwap_deviation_hard_block", "DELETED"),
        "neutral_upg_rsi":  cfg.neutral_upgrade.neutral_upgrade_min_rsi_score,
        "exit_guard":       cfg.position_exit_signal_guard.enabled,
        "exit_guard_bars":  cfg.position_exit_signal_guard.confirm_bars_required,
        "shrink_cap_1h":    "signal_1h" in cfg.signal_type_position_caps
                            .get("red_bar_shrinking", {})
                            .get("apply_to", []),
    }

    # 计算 hash（用于对比两次启动是否用了同一套配置）
    cfg_hash = hashlib.md5(
        json.dumps(snapshot, sort_keys=True).encode()
    ).hexdigest()[:8]

    self._log_info(
        f"[CONFIG_FINGERPRINT] hash={cfg_hash} | "
        + " | ".join(f"{k}={v}" for k, v in snapshot.items())
    )

# 插入位置：
#   1. __init__ 结束时调用一次
#   2. 每 20 个 cycle 调用一次（约每 5 小时）

# Binance Error: {'code': -2022, 'msg': 'ReduceOnly Order is rejected.'}

# -2022 的根本原因（三选一）：
# 原因 1: positionSide 设置错误

#   如果发出的是 positionSide=BOTH 或 SHORT → 被拒绝

#   → 0 仓位 + reduceOnly=True → 拒绝

# 说明全部平仓成功（通过其他路由成交了）
# 但 ExitGuard 发出的那一笔 market order 被 -2022 拒绝

# 结论：ExitGuard 发出的市价单被拒，但最终仓位还是被平掉了
# 最可能原因：ExitGuard 发出 reduceOnly market，
#              但 position size 已经被价格止损部分平掉，
#              导致请求量 > 剩余持仓

### B-2：修复代码

```python
# src/app/fund_flow_bot.py — _execute_close() 函数

def _execute_exit_guard_close(
    self,
    symbol:   str,
    ratio:    float,    # 1.0 = 全平，0.5 = 减半
    reason:   str,
) -> bool:
    """
    ExitGuard 专用平仓路由，修复 -2022 问题
    """
    # Step 1: 从交易所实时查询当前持仓量（不用本地缓存）
    position = self._fetch_live_position(symbol)   # 实时查询
    if position is None or position["qty"] <= 0:
        self._log_info(f"[ExitGuard] {symbol} 持仓已为 0，跳过平仓")
        return True   # 已经平了，视为成功

    current_qty  = position["qty"]
    close_qty    = current_qty * ratio
    position_side = position["positionSide"]   # "LONG" or "SHORT" or "BOTH"

    # Step 2: 计算平仓方向
    if position_side == "LONG" or position["side"] == "LONG":
        close_side = "SELL"
    else:
        close_side = "BUY"

    # Step 3: 构造订单（修复 positionSide 和 reduceOnly）
    order_params = {
        "symbol":       symbol,
        "side":         close_side,
        "type":         "MARKET",
        "quantity":     self._round_qty(close_qty, symbol),
        "reduceOnly":   True,
    }

    # Hedge mode 需要指定 positionSide
    if self._is_hedge_mode():
        order_params["positionSide"] = position_side
        order_params.pop("reduceOnly", None)   # hedge mode 不用 reduceOnly

    # Step 4: 下单，捕获 -2022 并降级处理
    try:
        result = self._place_order(**order_params)
        self._log_info(f"[ExitGuard] {symbol} close OK: {result}")
        return True

    except BinanceError as e:
        if e.code == -2022:
            # reduceOnly 被拒 → 尝试不带 reduceOnly 重发（风险：可能开反向仓）
            # 因此先二次确认持仓
            position_retry = self._fetch_live_position(symbol)
            if position_retry is None or position_retry["qty"] <= 0:
                self._log_info(f"[ExitGuard] {symbol} -2022 后持仓=0，忽略")
                return True

            # 持仓仍存在：记录告警，改用 close_position API
            self._log_warning(
                f"[ExitGuard] {symbol} -2022 ReduceOnly rejected, "
                f"retrying with closePosition=True"
            )
            order_params.pop("reduceOnly", None)
            order_params.pop("quantity",   None)
            order_params["closePosition"] = True   # 币安专用：直接全平该 positionSide
            result = self._place_order(**order_params)
            self._log_info(f"[ExitGuard] {symbol} close retry OK: {result}")
            return True

        # 其他错误：记录并返回 False
        self._log_error(f"[ExitGuard] {symbol} close FAILED: {e}")
        return False
```

```python
# src/fund_flow/macd_strategy_v2.py — 替换 vwap_hard_block 判断

def _check_vwap_deviation(
    self,
    dev_abs_pct: float,    # abs(price - vwap) / vwap
    atr_pct:     float,    # ATR / price
) -> str:
    """
    返回 "PASS" | "PENALTY" | "PROBE" | "BLOCK"
    """
    if atr_pct <= 0:
        # ATR 缺失：退化到原有 3% 固定门槛
        return "BLOCK" if dev_abs_pct > 0.03 else "PASS"

    ratio = dev_abs_pct / atr_pct

    if ratio < 1.5:   return "PASS"
    if ratio < 2.5:   return "PENALTY"
    if ratio < 4.0:   return "PROBE"     # ← 新增：不是 hard block
    return "BLOCK"                        # 4× ATR 以上才 block


# 在 vwap gate 处理中：
vwap_action = self._check_vwap_deviation(dev_abs_pct, atr_pct)

if vwap_action == "BLOCK":
    return self._neutral_signal(reason="vwap_hard_block", ...)

elif vwap_action == "PROBE":
    # 允许开仓，但强制 probe 仓位
    signal.max_portion_override = 0.06
    signal.vwap_probe_flag       = True
    # 不 return，继续后续评分

elif vwap_action == "PENALTY":
    signal.score_mult *= 0.88
    # 继续
```

### C-2：neutral_upgrade partial_confirm（明天，C-1 稳定后）

**为什么是第二步：释放最大的 HOLD 来源（36.9%）**

```python
# 当前 neutral_upgrade_gate 的核心问题
# 日志样例：BCHUSDT 4H=green_bar_shrinking, 1H=red_bar_growing
# 这是"4H 趋势减弱，1H 开始反弹"的早期入场信号
# 但因为 4H 没有明确方向，整个候选被 neutral_upgrade_gate 归零

# partial_confirm 路径：当 4H 是 shrink，1H 是 confirm 方向时，允许 probe

PARTIAL_CONFIRM_MAP = {
    # 4H 信号 → 允许的 1H 信号 → probe 方向
    "green_bar_shrinking": {
        "long_signals":  ["red_bar_growing", "flip_bullish"],   # 1H 多头确认
        "short_signals": ["green_bar_growing", "flip_bearish"],  # 1H 空头确认
    },
    "red_bar_shrinking": {
        "long_signals":  ["red_bar_growing", "flip_bullish"],
        "short_signals": ["green_bar_growing", "flip_bearish"],
    },
}

def _check_partial_confirm(
    self,
    signal_4h: str,
    signal_1h: str,
) -> dict | None:
    """
    返回 partial_confirm 结果，或 None（不适用）
    """
    mapping = PARTIAL_CONFIRM_MAP.get(signal_4h)
    if not mapping:
        return None

    if signal_1h in mapping["long_signals"]:
        return {"direction": "long",  "penalty": 0.85, "max_portion": 0.10}
    if signal_1h in mapping["short_signals"]:
        return {"direction": "short", "penalty": 0.85, "max_portion": 0.10}

    return None


# 在 neutral_upgrade_gate 处插入（在"无候选方向直接归零"之前）：
pc = self._check_partial_confirm(signal_4h, signal_1h)
if pc:
    # partial confirm 路径：允许开仓，但打折
    direction    = pc["direction"]
    score_mult   = pc["penalty"]          # 0.85
    max_portion  = pc["max_portion"]      # 0.10
    log_info(f"partial_confirm: {signal_4h}+{signal_1h}→{direction}")
    # 继续评分，不归零
else:
    # 原有逻辑：无法 upgrade → 归零
    return self._neutral_signal(reason="4H无明确方向", ...)
```
